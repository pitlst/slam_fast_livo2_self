安装cuda
```bash
wget https://developer.download.nvidia.com/compute/cuda/repos/ubuntu2404/x86_64/cuda-keyring_1.1-1_all.deb
sudo dpkg -i cuda-keyring_1.1-1_all.deb
sudo apt update
sudo apt install cuda-toolkit-12-8
echo 'export PATH=/usr/local/cuda-12.8/bin${PATH:+:${PATH}}' >> ~/.bashrc
echo 'export LD_LIBRARY_PATH=/usr/local/cuda-12.8/lib64${LD_LIBRARY_PATH:+:${LD_LIBRARY_PATH}}' >> ~/.bashrc
source ~/.bashrc
# 验证
nvcc -V
```

编译
```bash
cd /path/to/opencv
mkdir build && cd build 
cmake -D CMAKE_BUILD_TYPE=RELEASE -D INSTALL_PYTHON_EXAMPLES=OFF -D INSTALL_C_EXAMPLES=OFF -D WITH_TBB=ON -D WITH_CUDA=ON -D BUILD_opencv_cudacodec=OFF -D ENABLE_FAST_MATH=1 -D CUDA_FAST_MATH=1 -D WITH_CUBLAS=1 -D WITH_V4L=OFF -D WITH_GTK=ON -D WITH_OPENGL=OFF -D WITH_LIBV4L=ON -D WITH_QT=OFF -D WITH_GTK_2_X=ON -D WITH_GSTREAMER=ON -D OPENCV_GENERATE_PKGCONFIG=ON -D OPENCV_PC_FILE_NAME=opencv.pc -D OPENCV_ENABLE_NONFREE=ON -D CUDA_nppicom_LIBRARY=stdc++ -D OPENCV_EXTRA_MODULES_PATH=../../opencv_contrib/modules -D BUILD_EXAMPLES=OFF ..
```

获取并编译livox sdk2 
```bash
git clone https://github.com/Livox-SDK/Livox-SDK2.git
cd Livox-SDK2
mkdir build && cd build
make -j 24
```
在编译livox sdk2之前需要补充一件事，对于gcc13和gcc14不会自动引入 #include <cstdint>，所以需要手动修改

修改 sdk_core/logger_handler/file_manager.h和 sdk_core/comm/define.h 在开头加入 #include <cstdint>

然后编译


安装rerun用于可视化点云数据
```bash
# rerun cli
cargo install rerun-cli --locked
# rerun c++ sdk
git clone https://github.com/rerun-io/rerun.git
git checkout 0.32.0
mkdir build && cd build
cmake .. -DCMAKE_BUILD_TYPE=Release -DRERUN_DOWNLOAD_AND_BUILD_ARROW=ON
cmake --build . --parallel   --config Release --target rerun_sdk
cmake --install . --prefix ../../rerun_sdk_install
```
